================ FC_GENERAL  (FC 1) ================
Block title: CONTROL ON

--- Network 1: CONTROL ON   [LAD]
    "CONTROL ON FLAG" := "DI_EMERGENCY PB"

--- Network 2: MODE SELECTION   [LAD]
    "MANUAL MODE SELECTED" := "AlwaysTRUE" AND NOT "DI_AUTO MANUAL SELECTION"
    "AUTO MODE SELECTED" := "AlwaysTRUE" AND "DI_AUTO MANUAL SELECTION"

--- Network 3: FAULT RESET   [LAD]
    "FAULT RESET FLAG" := "DB_HMI"."FAULT RESET PB"

--- Network 4: (no title)   [LAD]
    (empty network)

================ FC_MANUAL  (FC 2) ================
Block title: MOTOR FWD /REV ON

--- Network 1: MOTOR FWD /REV ON   [LAD]
    "MANUAL CONVEYOR REV ON FLAG" := "CONTROL ON FLAG" AND "MANUAL MODE SELECTED" AND NOT "FAULT BIT" AND "DB_HMI"."MOTOR JOG REV PB"
    "MANUAL CONVEYOR FWD ON FLAG" := ("CONTROL ON FLAG" AND "MANUAL MODE SELECTED" AND NOT "FAULT BIT" AND "DB_HMI"."MOTOR JOG FWD PB" OR "CONTROL ON FLAG" AND "MANUAL MODE SELECTED" AND NOT "FAULT BIT" AND "DI_CYCLE START PB")

--- Network 2: (no title)   [LAD]
    (empty network)

================ FC_ALARMS  (FC 3) ================
Block title: EMERGENCY PB PRESSED FAULT (I0.4)

--- Network 1: EMERGENCY PB PRESSED FAULT (I0.4)   [LAD]
    SET   "DB_ALARMS"."EMERGENCY PB PRESSED FAULT"   when NOT "DI_EMERGENCY PB"

--- Network 2: CONVEYOR VFD FAULT (I0.5)   [LAD]
    SET   "DB_ALARMS"."VFD FAULT"   when "DI_CONVEYOR VFD FAULT"

--- Network 3: FAULT RESET   [LAD]
    MOVE 0 -> %DB2.DBW0   when "FAULT RESET FLAG"

--- Network 4: FAULT BIT   [LAD]
    "FAULT BIT" := [0 <> %DB2.DBW0 (Word)]
    "DB_ALARMS"."FAULT POP" := [0 <> %DB2.DBW0 (Word)]

    (empty network)

================ FC_AUTO  (FC 5) ================
Block title: AUTO ON

--- Network 1: AUTO ON   [LAD]
    "DB_AUTO".ACTIVE := ("CONTROL ON FLAG" AND "AUTO MODE SELECTED" AND NOT "FAULT BIT" AND "DI_CYCLE START PB" OR "CONTROL ON FLAG" AND "AUTO MODE SELECTED" AND NOT "FAULT BIT" AND "DB_AUTO".ACTIVE) AND "DI_CYCLE STOP PB"

--- Network 2: STOP DELAY TIME   [SCL]
    SCL:  "DB_AUTO"."STOP DELAY TIMER".PT := "DB_HMI"."STOP TIME" * 1000;

--- Network 3: CONVEYOR ON   [LAD]
    "DB_AUTO"."CONVEYOR ON" := ("AlwaysTRUE" AND "DB_AUTO".ACTIVE AND NOT "DB_AUTO"."CONVEYOR OFF2" OR "AlwaysTRUE" AND "DB_AUTO".ACTIVE AND "AlwaysTRUE") AND ["DB_CAMERA".DATA1 == 0 (Int)]
    SET   "DB_AUTO"."CONVEYOR OFF1"   when RISING_EDGE("AlwaysTRUE" AND NOT "AlwaysTRUE" AND "DB_AUTO".ACTIVE AND "DB_AUTO"."CONVEYOR ON" AND "DB_AUTO"."SENSOR CHECK DELAY".Q AND "DI_PRODUCT SENSOR", memory "DB_AUTO".P_TRG)
    TON "DB_AUTO"."SENSOR CHECK DELAY": IN = "AlwaysTRUE" AND NOT "AlwaysTRUE" AND "DB_AUTO".ACTIVE AND "DB_AUTO"."CONVEYOR ON", PT = T#500MS
    TON "DB_AUTO"."STOP DELAY TIMER": IN = "AlwaysTRUE" AND "DB_AUTO"."CONVEYOR OFF1", PT = "DB_AUTO"."STOP DELAY TIMER".PT
    RESET "DB_AUTO"."CONVEYOR OFF1"   when "AlwaysTRUE" AND "DB_AUTO"."STOP DELAY TIMER".Q
    MOVE 1 -> "DB_CAMERA".DATA1   when "AlwaysTRUE" AND "DB_AUTO"."STOP DELAY TIMER".Q
    SET   "DB_AUTO"."CONVEYOR OFF2"   when "AlwaysTRUE" AND "DB_AUTO"."STOP DELAY TIMER".Q

--- Network 4: CAMERA DONE   [LAD]
    "DB_AUTO"."CAMERA DONE" := "DB_AUTO"."CONVEYOR OFF2" AND ["DB_CAMERA".DATA1 == 0 (Int)]

--- Network 5: CONVEYOR OFF RESET   [LAD]
    RESET "DB_AUTO"."CONVEYOR OFF1"   when ("AlwaysTRUE" AND "DB_AUTO"."CONVEYOR OFF2" AND "DB_AUTO"."CAMERA DONE" OR "AlwaysTRUE" AND "DB_AUTO"."CONVEYOR OFF2" AND "DB_HMI"."CAMERA DONE" OR "AlwaysTRUE" AND NOT "DB_AUTO".ACTIVE)
    RESET "DB_AUTO"."CONVEYOR OFF2"   when ("AlwaysTRUE" AND "DB_AUTO"."CONVEYOR OFF2" AND "DB_AUTO"."CAMERA DONE" OR "AlwaysTRUE" AND "DB_AUTO"."CONVEYOR OFF2" AND "DB_HMI"."CAMERA DONE" OR "AlwaysTRUE" AND NOT "DB_AUTO".ACTIVE)

--- Network 6: AUTO STATUS   [LAD]
    MOVE 10 -> "DB_AUTO"."AUTO STATUS"   when "AlwaysTRUE" AND "DB_AUTO".ACTIVE AND NOT "DB_AUTO"."CONVEYOR OFF2"
    MOVE 20 -> "DB_AUTO"."AUTO STATUS"   when "AlwaysTRUE" AND NOT "DB_AUTO".ACTIVE
    MOVE 30 -> "DB_AUTO"."AUTO STATUS"   when "AlwaysTRUE" AND "DB_AUTO".ACTIVE AND "DB_AUTO"."CONVEYOR OFF2"

--- Network 7: (no title)   [LAD]
    (empty network)

================ FC_TCP_IP  (FC 6) ================
Block title: Connection definition

--- Network 1: Connection definition   [LAD]
    MOVE Local~PROFINET_interface_1 -> "DB_TCP_IP"."MY CONNECT DATA".InterfaceId   when TRUE
    MOVE 1 -> "DB_TCP_IP"."MY CONNECT DATA".ID   when TRUE
    MOVE 11 -> "DB_TCP_IP"."MY CONNECT DATA".ConnectionType   when TRUE
    "DB_TCP_IP"."MY CONNECT DATA".ActiveEstablished := NOT "AlwaysTRUE"
    MOVE 0 -> "DB_TCP_IP"."MY CONNECT DATA".RemoteAddress.ADDR[1]   when TRUE
    MOVE 0 -> "DB_TCP_IP"."MY CONNECT DATA".RemoteAddress.ADDR[2]   when TRUE
    MOVE 0 -> "DB_TCP_IP"."MY CONNECT DATA".RemoteAddress.ADDR[3]   when TRUE
    MOVE 0 -> "DB_TCP_IP"."MY CONNECT DATA".RemoteAddress.ADDR[4]   when TRUE
    MOVE 0 -> "DB_TCP_IP"."MY CONNECT DATA".RemotePort   when TRUE
    MOVE 502 -> "DB_TCP_IP"."MY CONNECT DATA".LocalPort   when TRUE

--- Network 2: MB SERVER   [LAD]
    MB_SERVER "MB_SERVER_DB" when TRUE: CONNECT = "DB_TCP_IP"."MY CONNECT DATA", DISCONNECT = "AlwaysFALSE", DR = (output) None, ERROR = (output) None, MB_HOLD_REG = P#DB5.DBX0.0 INT 10, NDR = (output) None, STATUS = (output) None

--- Network 3: ERROR CHECK   [LAD]
    "DB_TCP_IP"."NO OF DONE" := "DB_TCP_IP"."NO OF DONE" + 1   (UDInt) when ("MB_SERVER_DB".NDR OR "MB_SERVER_DB".DR)
    "DB_TCP_IP"."NO OF ERROR" := "DB_TCP_IP"."NO OF ERROR" + 1   (UDInt) when "MB_SERVER_DB".ERROR
    MOVE "MB_SERVER_DB".STATUS -> "DB_TCP_IP"."ERROR STATUS"   when "MB_SERVER_DB".ERROR
